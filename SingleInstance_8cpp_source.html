22bool SingleInstance::tryLock()
23{
24 // Primary mechanism: register a well-known D-Bus service on the session bus.
25 // On Wayland/Linux this both detects a previous instance and provides an
26 // activation path for future instances.
27 auto bus = QDBusConnection::sessionBus();
28 if (bus.isConnected()) {
29 if (bus.registerService(mo::core::constants::kDBusServiceName)) {
30 // Expose the "openFiles" method so that later instances can hand
31 // their command-line file arguments over to this instance.
32 if (!bus.registerObject(mo::core::constants::kDBusObjectPath, this,
33 QDBusConnection::ExportAllSlots)) {
34 mo::core::Logger::warning("Failed to register D-Bus object");
35 }
36 locked_ = true;
37 mo::core::Logger::info("Single instance acquired via D-Bus");
38 return true;
39 }
40 mo::core::Logger::warning(
41 "D-Bus service already registered; another instance is running");
42 return false;
43 }
44
45 // Fallback mechanism: a shared memory segment.
46 if (!sharedMemory_.create(1)) {
47 if (sharedMemory_.attach()) {
48 sharedMemory_.detach();
49 }
50 mo::core::Logger::warning(
51 "Another instance is already running (QSharedMemory fallback)");
52 return false;
53 }
54
55 locked_ = true;
56 return true;
57}

64bool SingleInstance::forwardToRunningInstance(const QStringList &files) const
65{
66 auto bus = QDBusConnection::sessionBus();
67 if (!bus.isConnected()) {
68 mo::core::Logger::warning("Cannot forward files: no D-Bus session bus");
69 return false;
70 }
71
72 auto message = QDBusMessage::createMethodCall(
73 mo::core::constants::kDBusServiceName,
74 mo::core::constants::kDBusObjectPath,
75 mo::core::constants::kDBusServiceName,
76 QStringLiteral("openFiles"));
77 message.setArguments({QVariant::fromValue(files)});
78 if (!bus.send(message)) {
79 mo::core::Logger::warning("Failed to forward files to the running instance");
80 return false;
81 }
82 mo::core::Logger::info("Forwarded files to the running instance");
83 return true;
84}
