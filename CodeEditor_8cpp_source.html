98int CodeEditor::lineNumberAreaWidth()
99{
100 if (!showLineNumbers_) {
101 return 0;
102 }
103 int digits = 1;
104 int max = qMax(1, blockCount());
105 while (max >= 10) {
106 max /= 10;
107 ++digits;
108 }
109 const int space = 3 + fontMetrics().horizontalAdvance(QLatin1Char('9')) * digits;
110 return space;
111}

139void CodeEditor::keyPressEvent(QKeyEvent *event)
140{
141 const int key = event->key();
142
143 // Column selection: Alt+Shift+arrows start or steer the rectangle.
144 if ((event->modifiers() & Qt::AltModifier) && (event->modifiers() & Qt::ShiftModifier)
145 && key >= Qt::Key_Left && key <= Qt::Key_Down) {
146 columnMoveCorner(key);
147 return;
148 }
149
150 if (columnActive_) {
151 switch (key) {
152 case Qt::Key_Escape:
153 cancelColumnSelection();
154 return;
155 case Qt::Key_Backspace:
156 columnDelete(false);
157 return;
158 case Qt::Key_Delete:
159 columnDelete(true);
160 return;
161 case Qt::Key_Tab:
162 columnInsertText(
163 QString(mo::core::Settings::instance().tabWidth(), QLatin1Char(' ')));
164 return;
165 case Qt::Key_Left:
166 case Qt::Key_Right:
167 case Qt::Key_Up:
168 case Qt::Key_Down:
169 // Plain arrow keys steer the multi-cursor group without leaving
170 // column mode (modified arrows fall through and cancel).
171 if (event->modifiers() == Qt::NoModifier) {
172 columnMoveCaret(key);
173 return;
174 }
175 break;
176 default:
177 break;
178 }
179 if (key == Qt::Key_C || key == Qt::Key_X || key == Qt::Key_V) {
180 if ((event->modifiers() & Qt::ControlModifier)) {
181 if (key == Qt::Key_C) {
182 columnCopy();
183 } else if (key == Qt::Key_X) {
184 columnCut();
185 } else {
186 columnPaste();
187 }
188 return;
189 }
190 }
191 // Plain text input (no Ctrl/Alt/Meta modifiers).
192 if (event->text().size() == 1
193 && !(event->modifiers()
194 & (Qt::ControlModifier | Qt::AltModifier | Qt::MetaModifier))) {
195 columnInsertText(event->text());
196 return;
197 }
198 // Modifier-only presses must not cancel the selection: they arrive
199 // as separate key events while composing the next Alt+Shift+Arrow
200 // chord (users press and release the modifiers for every key).
201 if (key == Qt::Key_Shift || key == Qt::Key_Control || key == Qt::Key_Alt
202 || key == Qt::Key_Meta || key == Qt::Key_AltGr) {
203 QPlainTextEdit::keyPressEvent(event);
204 return;
205 }
206 // Anything else cancels the column selection and falls through to
207 // the regular handling below.
208 cancelColumnSelection();
209 }
210
211 // Auto-indent: when pressing Enter, copy leading whitespace from current line.
212 if (mo::core::Settings::instance().autoIndent()
213 && event->key() == Qt::Key_Return && !(event->modifiers() & Qt::ShiftModifier)) {
214 auto cursor = textCursor();
215 const QString lineText = cursor.block().text();
216 QString indent;
217 for (const auto ch : lineText) {
218 if (ch == QLatin1Char(' ') || ch == QLatin1Char('\t')) {
219 indent += ch;
220 } else {
221 break;
222 }
223 }
224 QPlainTextEdit::keyPressEvent(event);
225 if (!indent.isEmpty()) {
226 cursor = textCursor();
227 cursor.insertText(indent);
228 }
229 return;
230 }
231
232 // Tab key: insert spaces according to tabWidth setting.
233 if (event->key() == Qt::Key_Tab && !(event->modifiers() & Qt::ControlModifier)) {
234 const int tw = mo::core::Settings::instance().tabWidth();
235 insertPlainText(QString(tw, QLatin1Char(' ')));
236 return;
237 }
238
239 QPlainTextEdit::keyPressEvent(event);
240}

248void CodeEditor::paintEvent(QPaintEvent *event)
249{
250 QPlainTextEdit::paintEvent(event);
251
252 // A zero-width column selection is drawn as a thin caret on every row.
253 if (!columnActive_ || columnRectLeft() != columnRectRight()) {
254 return;
255 }
256 QPainter painter(viewport());
257 const QColor color = palette().color(QPalette::Highlight);
258 for (int line = columnFirstLine(); line <= columnLastLine(); ++line) {
259 const auto block = document()->findBlockByNumber(line);
260 if (!block.isValid()) {
261 break;
262 }
263 QTextCursor cursor(document());
264 cursor.setPosition(block.position() + qMin(columnRectLeft(), block.text().size()));
265 const QRect rect = cursorRect(cursor);
266 painter.fillRect(QRect(rect.left(), rect.top(), 2, rect.height()), color);
267 }
268}

270void CodeEditor::highlightCurrentLine()
271{
272 QList<QTextEdit::ExtraSelection> extraSelections;
273 if (!isReadOnly()) {
274 QTextEdit::ExtraSelection selection;
275 QColor lineColor;
276 if (highlightingTheme_.isValid()) {
277 lineColor = QColor::fromRgba(highlightingTheme_.editorColor(
278 KSyntaxHighlighting::Theme::EditorColorRole::CurrentLine));
279 }
280 if (!lineColor.isValid()) {
281 lineColor = palette().color(QPalette::AlternateBase);
282 }
283 selection.format.setBackground(lineColor);
284 selection.format.setProperty(QTextFormat::FullWidthSelection, true);
285 selection.cursor = textCursor();
286 selection.cursor.clearSelection();
287 extraSelections.append(selection);
288 }
289
290 // Column (rectangular) selection: one highlighted range per row, clamped
291 // to each line's length.
292 if (columnActive_) {
293 const QColor selColor = palette().color(QPalette::Highlight);
294 const int left = columnRectLeft();
295 const int right = columnRectRight();
296 for (int line = columnFirstLine(); line <= columnLastLine(); ++line) {
297 const auto block = document()->findBlockByNumber(line);
298 if (!block.isValid()) {
299 break;
300 }
301 const int len = block.text().size();
302 const int start = qMin(left, len);
303 const int end = qMin(right, len);
304 if (end <= start) {
305 continue;
306 }
307 QTextEdit::ExtraSelection sel;
308 sel.format.setBackground(selColor);
309 sel.cursor = QTextCursor(document());
310 sel.cursor.setPosition(block.position() + start);
311 sel.cursor.setPosition(block.position() + end, QTextCursor::KeepAnchor);
312 extraSelections.append(sel);
313 }
314 }
315 setExtraSelections(extraSelections);
316}

318void CodeEditor::lineNumberAreaPaintEvent(QPaintEvent *event)
319{
320 QPainter painter(lineNumberArea_);
321
322 QColor bgColor;
323 QColor fgColor;
324 if (highlightingTheme_.isValid()) {
325 bgColor = QColor::fromRgba(highlightingTheme_.editorColor(
326 KSyntaxHighlighting::Theme::EditorColorRole::IconBorder));
327 fgColor = QColor::fromRgba(highlightingTheme_.editorColor(
328 KSyntaxHighlighting::Theme::EditorColorRole::LineNumbers));
329 }
330 if (!bgColor.isValid()) {
331 bgColor = palette().color(QPalette::Window);
332 }
333 if (!fgColor.isValid()) {
334 fgColor = palette().color(QPalette::BrightText);
335 if (!fgColor.isValid()) {
336 fgColor = palette().color(QPalette::Text);
337 }
338 }
339
340 painter.fillRect(event->rect(), bgColor);
341
342 QTextBlock block = firstVisibleBlock();
343 int blockNumber = block.blockNumber();
344 int top = qRound(blockBoundingGeometry(block).translated(contentOffset()).top());
345 int bottom = top + qRound(blockBoundingRect(block).height());
346
347 painter.setPen(fgColor);
348 while (block.isValid() && top <= event->rect().bottom()) {
349 if (block.isVisible() && bottom >= event->rect().top()) {
350 const QString number = QString::number(blockNumber + 1);
351 painter.drawText(0, top, lineNumberArea_->width() - 2,
352 fontMetrics().height(), Qt::AlignRight, number);
353 }
354 block = block.next();
355 top = bottom;
356 bottom = top + qRound(blockBoundingRect(block).height());
357 ++blockNumber;
358 }
359}

361bool CodeEditor::loadFile(const QString &path)
362{
363 QFile file(path);
364 if (!file.open(QIODevice::ReadOnly | QIODevice::Text)) {
365 mo::core::Logger::warning("Failed to open: " + path);
366 return false;
367 }
368 QTextStream in(&file);
369 in.setEncoding(QStringConverter::Utf8);
370 setPlainText(in.readAll());
371 file.close();
372
373 filePath_ = path;
374 setLanguage(QFileInfo(path).suffix());
375 cancelColumnSelection();
376 document()->setModified(false);
377 updateTabTitle();
378 return true;
379}

389bool CodeEditor::saveAs(const QString &path)

392 if (!file.open(QIODevice::WriteOnly | QIODevice::Text)) {
393 return false;
394 }
395 QTextStream out(&file);
396 out.setEncoding(QStringConverter::Utf8);
397 out << toPlainText();
398 out.flush();
399 if (file.commit()) {
400 filePath_ = path;
401 document()->setModified(false);
402 updateTabTitle();
403 return true;
404 }
405 return false;

529{

552void CodeEditor::columnMoveCorner(int key)
553{
554 if (!columnActive_) {
555 const auto cursor = textCursor();
556 columnAnchor_ = {cursor.blockNumber(), cursor.columnNumber()};
557 columnCorner_ = columnAnchor_;
558 columnActive_ = true;
559 }
560 switch (key) {
561 case Qt::Key_Up:
562 columnCorner_.line = qMax(0, columnCorner_.line - 1);
563 break;
564 case Qt::Key_Down:
565 columnCorner_.line = qMin(blockCount() - 1, columnCorner_.line + 1);
566 break;
567 case Qt::Key_Left:
568 columnCorner_.column = qMax(0, columnCorner_.column - 1);
569 break;
570 case Qt::Key_Right:
571 ++columnCorner_.column;
572 break;
573 default:
574 return;
575 }
576 columnUpdateCaret();
577 highlightCurrentLine();
578 viewport()->update();
579}

581void CodeEditor::columnMoveCaret(int key)
582{
583 // Plain arrow keys steer the whole multi-cursor group without leaving
584 // column mode: Up/Down shift every row, Left/Right shift the column.
585 // A selection with width collapses first, the way a plain caret would
586 // jump to the edge of its selection before moving further.
587 const int left = columnRectLeft();
588 const int right = columnRectRight();
589 const bool hadWidth = left != right;
590
591 switch (key) {
592 case Qt::Key_Left:
593 columnAnchor_.column = columnCorner_.column =
594 hadWidth ? left : qMax(left - 1, 0);
595 break;
596 case Qt::Key_Right:
597 columnAnchor_.column = columnCorner_.column = hadWidth ? right : right + 1;
598 break;
599 case Qt::Key_Up:
600 case Qt::Key_Down: {
601 if (hadWidth) {
602 // The selection collapses to the active corner's column.
603 columnAnchor_.column = columnCorner_.column;
604 }
605 const int first = qMin(columnAnchor_.line, columnCorner_.line);
606 const int last = qMax(columnAnchor_.line, columnCorner_.line);
607 const int bottom = document()->blockCount() - 1;
608 if (key == Qt::Key_Up && first > 0) {
609 --columnAnchor_.line;
610 --columnCorner_.line;
611 } else if (key == Qt::Key_Down && last < bottom) {
612 ++columnAnchor_.line;
613 ++columnCorner_.line;
614 }
615 break;
616 }
617 default:
618 return;
619 }
620 columnUpdateCaret();
621 highlightCurrentLine();
622 viewport()->update();
623}

636void CodeEditor::columnInsertText(const QString &text)
637{
638 if (!columnActive_ || text.isEmpty()) {
639 return;
640 }
641 const int left = columnRectLeft();
642 const int right = columnRectRight();
643
644 QTextCursor edit(textCursor());
645 edit.beginEditBlock();
646 for (int line = columnLastLine(); line >= columnFirstLine(); --line) {
647 const auto block = document()->findBlockByNumber(line);
648 if (!block.isValid()) {
649 continue;
650 }
651 const int len = block.text().size();
652 const int start = qMin(left, len);
653 QTextCursor cursor(document());
654 cursor.setPosition(block.position() + start);
655 if (right > left) {
656 const int end = qMin(right, len);
657 if (end > start) {
658 cursor.setPosition(block.position() + end, QTextCursor::KeepAnchor);
659 cursor.removeSelectedText();
660 }
661 }
662 cursor.insertText(text);
663 }
664 edit.endEditBlock();
665
666 // Keep the column selection alive after the inserted text.
667 columnAnchor_.column = columnCorner_.column = left + text.size();
668 columnUpdateCaret();
669 highlightCurrentLine();
670 viewport()->update();
671}

673void CodeEditor::columnDelete(bool forward)
674{
675 if (!columnActive_) {
676 return;
677 }
678 if (columnRectLeft() != columnRectRight()) {
679 columnDeleteBlock();
680 return;
681 }
682
683 const int col = columnRectLeft();
684 QTextCursor edit(textCursor());
685 edit.beginEditBlock();
686 for (int line = columnLastLine(); line >= columnFirstLine(); --line) {
687 const auto block = document()->findBlockByNumber(line);
688 if (!block.isValid()) {
689 continue;
690 }
691 const int len = block.text().size();
692 const int pos = qMin(col, len);
693 QTextCursor cursor(document());
694 if (forward) {
695 if (pos >= len) {
696 continue; // nothing to delete after end of line
697 }
698 cursor.setPosition(block.position() + pos);
699 cursor.deleteChar();
700 } else {
701 if (pos == 0) {
702 continue; // nothing to delete before start of line
703 }
704 cursor.setPosition(block.position() + pos - 1);
705 cursor.deleteChar();
706 }
707 }
708 edit.endEditBlock();
709
710 // The zero-width column selection persists, like multi-cursor editing.
711 // A backward delete also shifts the caret column left so consecutive
712 // Backspaces keep deleting the character left of the caret, matching
713 // single-caret behaviour.
714 if (!forward && col > 0) {
715 columnAnchor_.column = columnCorner_.column = col - 1;
716 }
717 columnUpdateCaret();
718 highlightCurrentLine();
719 viewport()->update();
720}

722void CodeEditor::columnDeleteBlock()
723{
724 const int left = columnRectLeft();
725 const int right = columnRectRight();
726
727 QTextCursor edit(textCursor());
728 edit.beginEditBlock();
729 for (int line = columnLastLine(); line >= columnFirstLine(); --line) {
730 const auto block = document()->findBlockByNumber(line);
731 if (!block.isValid()) {
732 continue;
733 }
734 const int len = block.text().size();
735 const int start = qMin(left, len);
736 const int end = qMin(right, len);
737 if (end <= start) {
738 continue;
739 }
740 QTextCursor cursor(document());
741 cursor.setPosition(block.position() + start);
742 cursor.setPosition(block.position() + end, QTextCursor::KeepAnchor);
743 cursor.removeSelectedText();
744 }
745 edit.endEditBlock();
746
747 // Keep the multi-cursor alive: the rectangle collapses to a zero-width
748 // column at its left edge, ready for further typing or deletion. Only
749 // Esc, a mouse click, or a plain key exits column mode.
750 columnAnchor_.column = columnCorner_.column = left;
751 columnUpdateCaret();
752 highlightCurrentLine();
753 viewport()->update();
754}

756void CodeEditor::columnCopy()
757{
758 if (!columnActive_ || columnRectLeft() == columnRectRight()) {
759 return;
760 }
761 const int left = columnRectLeft();
762 const int right = columnRectRight();
763 QString text;
764 for (int line = columnFirstLine(); line <= columnLastLine(); ++line) {
765 const auto block = document()->findBlockByNumber(line);
766 if (!block.isValid()) {
767 break;
768 }
769 const int len = block.text().size();
770 const int start = qMin(left, len);
771 const int end = qMin(right, len);
772 if (!text.isEmpty()) {
773 text += QLatin1Char('\n');
774 }
775 if (end > start) {
776 text += block.text().mid(start, end - start);
777 }
778 }
779 QGuiApplication::clipboard()->setText(text);
780}

791void CodeEditor::columnPaste()
792{
793 if (!columnActive_) {
794 paste();
795 return;
796 }
797 const QStringList lines = QGuiApplication::clipboard()->text().split(QLatin1Char('\n'));
798 const int rows = columnLastLine() - columnFirstLine() + 1;
799 if (lines.size() == 1 && rows > 1) {
800 // Single-line clipboard text acts like multi-cursor typing: insert it
801 // on every selected row (replacing the rectangle when it has width).
802 columnInsertText(lines.constFirst());
803 return;
804 }
805 if (lines.size() != rows) {
806 // Line counts do not match: fall back to a normal paste at the caret.
807 cancelColumnSelection();
808 paste();
809 return;
810 }
811
812 const int left = columnRectLeft();
813 const int right = columnRectRight();
814 QTextCursor edit(textCursor());
815 edit.beginEditBlock();
816 for (int i = lines.size() - 1; i >= 0; --i) {
817 const auto block = document()->findBlockByNumber(columnFirstLine() + i);
818 if (!block.isValid()) {
819 continue;
820 }
821 const int len = block.text().size();
822 const int start = qMin(left, len);
823 QTextCursor cursor(document());
824 cursor.setPosition(block.position() + start);
825 if (right > left) {
826 const int end = qMin(right, len);
827 if (end > start) {
828 cursor.setPosition(block.position() + end, QTextCursor::KeepAnchor);
829 cursor.removeSelectedText();
830 }
831 }
832 cursor.insertText(lines.at(i));
833 }
834 edit.endEditBlock();
835
836 // Keep the column selection alive after the pasted text; the caret
837 // column advances past the longest pasted line.
838 int width = 0;
839 for (const QString &line : lines) {
840 width = qMax(width, line.size());
841 }
842 columnAnchor_.column = columnCorner_.column = left + width;
843 columnUpdateCaret();
844 highlightCurrentLine();
845 viewport()->update();
846}
