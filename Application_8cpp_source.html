23{

30 initLogging();
31 loadTranslation();
32 parseCommandLine();
33
34 singleInstance_ = std::make_unique<SingleInstance>(this);
35 if (!singleInstance_->tryLock()) {
36 mo::core::Logger::warning("Another instance is already running; exiting");
37 if (!filesToOpen_.isEmpty()) {
38 // Hand the file arguments over to the running instance.
39 singleInstance_->forwardToRunningInstance(filesToOpen_);
40 }
41 return;
42 }
43
44 // A second instance may hand over its file arguments while we run.
45 connect(singleInstance_.get(), &SingleInstance::openFilesRequested, this,
46 [this](const QStringList &files) {
47 if (!mainWindow_) {
48 return;
49 }
50 for (const auto &file : files) {
51 mainWindow_->openFile(file);
52 }
53 mainWindow_->show();
54 mainWindow_->raise();
55 mainWindow_->activateWindow();
56 });
57
58 mo::core::Settings::instance().load();
59
60 mainWindow_ = std::make_unique<mo::ui::MainWindow>(filesToOpen_);
61 mainWindow_->show();
62}

66int Application::run()
67{
68 if (!mainWindow_) {
69 // Another instance already holds the lock; quit immediately.
70 return 0;
71 }
72 connect(this, &QCoreApplication::aboutToQuit, this, &Application::saveSession);

84void Application::loadTranslation()
85{
86 // Qt's own translation (dialogs, standard shortcuts).
87 static QTranslator qtTranslator;
88 if (qtTranslator.load(QLocale::system(), QStringLiteral("qt"), QStringLiteral("_"),
89 QLibraryInfo::path(QLibraryInfo::TranslationsPath))) {
90 installTranslator(&qtTranslator);
91 }
92
93 // Application translation; load zh_CN resource when locale is Chinese.
94 static QTranslator appTranslator;
95 const auto locale = QLocale().name();
96 if (locale.startsWith(QStringLiteral("zh"))) {
97 if (appTranslator.load(QStringLiteral(":/i18n/mo_zh_CN.qm"))) {
98 installTranslator(&appTranslator);
99 } else {
100 mo::core::Logger::warning("Failed to load zh_CN translation resource");
101 }
102 }
103}

110void Application::parseCommandLine()
111{
112 QCommandLineParser parser;
113 parser.setApplicationDescription(
114 tr("A fast and lightweight multi-tab text editor"));
115 parser.addHelpOption();
116 parser.addVersionOption();
117 parser.addPositionalArgument(QStringLiteral("files"),
118 tr("Files to open, each in its own tab."),
119 QStringLiteral("[files...]"));
120 parser.process(*this);
121
122 // Normalize to absolute paths so that later session restore and recent
123 // files work regardless of the working directory.
124 for (const auto &file : parser.positionalArguments()) {
125 filesToOpen_.append(QFileInfo(file).absoluteFilePath());
126 }
127}
